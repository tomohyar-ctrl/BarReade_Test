async function startCamera() { try { log("カメラ開始処理を開始"); // Safari で確実に起動する背面カメラ指定（exact を使わない） currentStream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: "environment" // ← これが最も安定 } }); video.srcObject = currentStream; await video.play(); // ZXing デコード開始（deviceId を使わない） await codeReader.decodeFromVideoElement( video, (result, err) => { if (result) { const text = result.getText(); if (text !== lastResultText) { lastResultText = text; resultDiv.textContent = `読み取り結果：${text}`; log(`コード検出: ${text}`); // ピッ音 beep.play(); } } else if (err && !(err instanceof ZXing.NotFoundException)) { log(`デコードエラー: ${err}`); } } ); startBtn.disabled = true; stopBtn.disabled = false; log("背面カメラで開始完了"); } catch (e) { console.error(e); log(`カメラ起動エラー: ${e}`); alert("カメラの起動に失敗しました。HTTPSでアクセスしているか、権限を確認してください。"); } }
